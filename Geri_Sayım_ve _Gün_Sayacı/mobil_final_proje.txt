enum SortOption { listOrder, closestFirst, furthestFirst, newestFirst, oldestFirst, nameAZ, nameZA }

class CountdownModel {
  final String id;
  String title;
  DateTime targetDateTime;
  int themeIndex;
  final DateTime createdAt;

  CountdownModel({
    required this.id,
    required this.title,
    required this.targetDateTime,
    required this.themeIndex,
    required this.createdAt,
  });


  Map<String, dynamic> toJson() => {
    'id': id,
    'title': title,
    'targetDateTime': targetDateTime.toIso8601String(),
    'themeIndex': themeIndex,
    'createdAt': createdAt.toIso8601String(),
  };


  factory CountdownModel.fromJson(Map<String, dynamic> json) => CountdownModel(
    id: json['id'],
    title: json['title'],
    targetDateTime: DateTime.parse(json['targetDateTime']),
    themeIndex: json['themeIndex'],
    createdAt: DateTime.parse(json['createdAt']),
  );


  List<Color> get colors {
    switch (themeIndex) {
      case 1: return [const Color(0xFF4EA8DE), const Color(0xFF56CFE1)];
      case 2: return [const Color(0xFFE056FD), const Color(0xFFED4C67)];
      case 3: return [const Color(0xFFF9A825), const Color(0xFFFBC02D)];
      case 4: return [const Color(0xFF90BE6D), const Color(0xFF43AA8B)];
      case 5: return [const Color(0xFFF94144), const Color(0xFFF8961E)];
      case 6: return [const Color(0xFF9D4EDD), const Color(0xFF7B2CBF)];
      case 7: return [const Color(0xFF48CAE4), const Color(0xFF0077B6)];
      case 8: return [const Color(0xFF4F5D75), const Color(0xFF2D3142)];
      default: return [const Color(0xFF1EA3B2), const Color(0xFF147A85)];
    }
  }


  String get themeName {
    switch (themeIndex) {
      case 1: return 'Tema 1 · Okyanus';
      case 2: return 'Tema 2 · Gün Batımı';
      case 3: return 'Tema 3 · Şafak';
      case 4: return 'Tema 4 · Çayır';
      case 5: return 'Tema 5 · Mercan';
      case 6: return 'Tema 6 · Mor Tan';
      case 7: return 'Tema 7 · Buz';
      case 8: return 'Tema 8 · Kömür';
      default: return 'Tema $themeIndex';
    }
  }
}


class HomeScreen extends StatefulWidget {
  final bool isDarkMode;
  final VoidCallback onToggleTheme;

  const HomeScreen({super.key, required this.isDarkMode, required this.onToggleTheme});

  @override
  State<HomeScreen> createState() => _HomeScreenState();
}

class _HomeScreenState extends State<HomeScreen> {
  final List<CountdownModel> _countdowns = [];

  String _searchQuery = '';
  SortOption _currentSort = SortOption.listOrder;
  late Timer _globalTimer;

  final AudioPlayer _audioPlayer = AudioPlayer();
  final Set<String> _playedCountdowns = {};

  @override
  void initState() {
    super.initState();


    _audioPlayer.setReleaseMode(ReleaseMode.release);

    _loadCountdownsFromPrefs();


    _globalTimer = Timer.periodic(const Duration(seconds: 1), (timer) {
      if (mounted) {
        setState(() {});
        _checkCountdownsAndPlaySound();
      }
    });
  }


  Future<void> _loadCountdownsFromPrefs() async {
    try {
      final prefs = await SharedPreferences.getInstance();
      final String? savedData = prefs.getString('countdowns');
      if (savedData != null) {
        final List<dynamic> decoded = jsonDecode(savedData);
        setState(() {
          _countdowns.clear();
          _countdowns.addAll(decoded.map((item) => CountdownModel.fromJson(item)).toList());
        });
      }
    } catch (e) {
      debugPrint("Hafızadan veri çekilirken hata oluştu: $e");
    }
  }


  Future<void> _saveCountdownsToPrefs() async {
    final prefs = await SharedPreferences.getInstance();
    final String encoded = jsonEncode(_countdowns.map((item) => item.toJson()).toList());
    await prefs.setString('countdowns', encoded);
  }


  void _checkCountdownsAndPlaySound() {
    final now = DateTime.now();
    for (var countdown in _countdowns) {
      if (countdown.targetDateTime.isBefore(now) && !_playedCountdowns.contains(countdown.id)) {
        _playedCountdowns.add(countdown.id);
        _playSound();
      }
    }
  }


  Future<void> _playSound() async {
    try {
      await _audioPlayer.play(AssetSource('sounds/notification.mp3'));
    } catch (e) {
      debugPrint("Ses çalınırken hata oluştu: $e");
    }
  }

  @override
  void dispose() {
    _globalTimer.cancel();
    _audioPlayer.dispose();
    super.dispose();
  }


  List<CountdownModel> get _processedCountdowns {
    List<CountdownModel> list = _countdowns.where((item) {
      final title = item.title.isEmpty ? 'İsimsiz' : item.title;
      return title.toLowerCase().contains(_searchQuery.toLowerCase());
    }).toList();

    switch (_currentSort) {
      case SortOption.closestFirst:
        list.sort((a, b) => a.targetDateTime.compareTo(b.targetDateTime));
        break;
      case SortOption.furthestFirst:
        list.sort((a, b) => b.targetDateTime.compareTo(a.targetDateTime));
        break;
      case SortOption.newestFirst:
        list.sort((a, b) => b.createdAt.compareTo(a.createdAt));
        break;
      case SortOption.oldestFirst:
        list.sort((a, b) => a.createdAt.compareTo(b.createdAt));
        break;
      case SortOption.nameAZ:
        list.sort((a, b) => a.title.toLowerCase().compareTo(b.title.toLowerCase()));
        break;
      case SortOption.nameZA:
        list.sort((a, b) => b.title.toLowerCase().compareTo(a.title.toLowerCase()));
        break;
      case SortOption.listOrder:
        break;
    }
    return list;
  }


  void _openSortMenu() {
    showModalBottomSheet(
      context: context,
      shape: const RoundedRectangleBorder(borderRadius: BorderRadius.vertical(top: Radius.circular(16))),
      builder: (context) {
        return SafeArea(
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              const Padding(
                padding: EdgeInsets.all(16.0),
                child: Text('Şuna göre sırala', style: TextStyle(fontSize: 16, fontWeight: FontWeight.bold)),
              ),
              const Divider(height: 1),
              ListTile(
                leading: const Icon(Icons.sort),
                title: const Text('Listedeki sıra'),
                trailing: _currentSort == SortOption.listOrder ? const Icon(Icons.check, color: Colors.teal) : null,
                onTap: () => setState(() { _currentSort = SortOption.listOrder; Navigator.pop(context); }),
              ),
              ListTile(
                leading: const Icon(Icons.hourglass_top),
                title: const Text('Yaklaşanlar en başta'),
                trailing: _currentSort == SortOption.closestFirst ? const Icon(Icons.check, color: Colors.teal) : null,
                onTap: () => setState(() { _currentSort = SortOption.closestFirst; Navigator.pop(context); }),
              ),
              ListTile(
                leading: const Icon(Icons.hourglass_bottom),
                title: const Text('Yaklaşanlar en sonda'),
                trailing: _currentSort == SortOption.furthestFirst ? const Icon(Icons.check, color: Colors.teal) : null,
                onTap: () => setState(() { _currentSort = SortOption.furthestFirst; Navigator.pop(context); }),
              ),
              ListTile(
                leading: const Icon(Icons.history),
                title: const Text('Yeniden eskiye'),
                trailing: _currentSort == SortOption.newestFirst ? const Icon(Icons.check, color: Colors.teal) : null,
                onTap: () => setState(() { _currentSort = SortOption.newestFirst; Navigator.pop(context); }),
              ),
              ListTile(
                leading: const Icon(Icons.update),
                title: const Text('Eskiden yeniye'),
                trailing: _currentSort == SortOption.oldestFirst ? const Icon(Icons.check, color: Colors.teal) : null,
                onTap: () => setState(() { _currentSort = SortOption.oldestFirst; Navigator.pop(context); }),
              ),
              ListTile(
                leading: const Icon(Icons.sort_by_alpha),
                title: const Text('İsim (A-Z)'),
                trailing: _currentSort == SortOption.nameAZ ? const Icon(Icons.check, color: Colors.teal) : null,
                onTap: () => setState(() { _currentSort = SortOption.nameAZ; Navigator.pop(context); }),
              ),
              ListTile(
                leading: const Icon(Icons.sort_by_alpha),
                title: const Text('İsim (Z-A)'),
                trailing: _currentSort == SortOption.nameZA ? const Icon(Icons.check, color: Colors.teal) : null,
                onTap: () => setState(() { _currentSort = SortOption.nameZA; Navigator.pop(context); }),
              ),
            ],
          ),
        );
      },
    );
  }

  @override
  Widget build(BuildContext context) {
    final displayList = _processedCountdowns;

    return Scaffold(
      appBar: AppBar(
        title: Container(
          height: 40,
          decoration: BoxDecoration(
            color: Colors.white.withValues(alpha: 0.18),
            borderRadius: BorderRadius.circular(8),
          ),
          child: TextField(
            onChanged: (value) => setState(() => _searchQuery = value),
            style: const TextStyle(color: Colors.white),
            decoration: const InputDecoration(
              hintText: 'Arayın...',
              hintStyle: TextStyle(color: Colors.white60, fontSize: 14),
              prefixIcon: Icon(Icons.search, color: Colors.white60, size: 20),
              border: InputBorder.none,
              contentPadding: EdgeInsets.symmetric(vertical: 8),
            ),
          ),
        ),
        actions: [
          IconButton(icon: const Icon(Icons.tune), onPressed: _openSortMenu),
          IconButton(
            icon: Icon(widget.isDarkMode ? Icons.wb_sunny : Icons.nightlight_round),
            onPressed: widget.onToggleTheme,
          ),
        ],
      ),
      body: displayList.isEmpty
          ? Center(
        child: Padding(
          padding: const EdgeInsets.symmetric(horizontal: 32.0),
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              Icon(
                Icons.access_time,
                size: 48,
                color: widget.isDarkMode ? Colors.white54 : const Color(0xFF4F5D75),
              ),
              const SizedBox(height: 16),
              Text(
                'Henüz sayaç yok',
                style: TextStyle(
                  fontSize: 16,
                  fontWeight: FontWeight.bold,
                  color: widget.isDarkMode ? Colors.white : Colors.black87,
                ),
              ),
              const SizedBox(height: 8),
              Text(
                'Yeni bir geri sayım eklemek için + butonuna dokunun',
                textAlign: TextAlign.center,
                style: TextStyle(
                  fontSize: 12,
                  color: widget.isDarkMode ? Colors.white60 : Colors.grey[600],
                ),
              ),
            ],
          ),
        ),
      )

          : ListView.builder(
        padding: const EdgeInsets.all(14),
        itemCount: displayList.length,
        itemBuilder: (context, index) {
          final item = displayList[index];
          return GestureDetector(
            onTap: () async {
              final result = await Navigator.push(
                context,
                MaterialPageRoute(builder: (context) => ManageCountdownScreen(model: item)),
              );
              if (result == 'delete') {
                setState(() {
                  _countdowns.removeWhere((e) => e.id == item.id);
                  _playedCountdowns.remove(item.id);
                });
                _saveCountdownsToPrefs();
              } else if (result == 'save') {
                setState(() {
                  if (item.targetDateTime.isAfter(DateTime.now())) {
                    _playedCountdowns.remove(item.id);
                  }
                });
                _saveCountdownsToPrefs();

                if (context.mounted) {
                  final name = item.title.isEmpty ? 'İsimsiz' : item.title;
                  ScaffoldMessenger.of(context).showSnackBar(
                    SnackBar(
                      content: Text('"$name" başarıyla kaydedildi.'),
                      behavior: SnackBarBehavior.floating,
                      backgroundColor: const Color(0xFF1EA3B2),
                    ),
                  );
                }
              }
            },
            child: CountdownCard(model: item),
          );
        },
      ),
      floatingActionButton: FloatingActionButton(
        backgroundColor: const Color(0xFF1EA3B2),
        foregroundColor: Colors.white,
        shape: const CircleBorder(),
        child: const Icon(Icons.add),
        onPressed: () async {
          final result = await Navigator.push(
            context,
            MaterialPageRoute(builder: (context) => const ManageCountdownScreen()),
          );
          if (result != null && result is CountdownModel) {
            setState(() => _countdowns.add(result));
            _saveCountdownsToPrefs();

            if (context.mounted) {
              final name = result.title.isEmpty ? 'İsimsiz' : result.title;
              ScaffoldMessenger.of(context).showSnackBar(
                SnackBar(
                  content: Text('"$name" başarıyla kaydedildi.'),
                  behavior: SnackBarBehavior.floating,
                  backgroundColor: const Color(0xFF1EA3B2),
                ),
              );
            }
          }
        },
      ),
    );
  }
}


class CountdownCard extends StatelessWidget {
  final CountdownModel model;
  const CountdownCard({super.key, required this.model});

  @override
  Widget build(BuildContext context) {
    final now = DateTime.now();
    final diff = model.targetDateTime.difference(now);
    final bool isFinished = diff.isNegative;

    int days = 0;
    int hours = 0;
    int minutes = 0;
    int seconds = 0;

    if (!isFinished) {
      days = diff.inDays;
      hours = diff.inHours % 24;
      minutes = diff.inMinutes % 60;
      seconds = diff.inSeconds % 60;
    }

    String twoDigits(int n) => n.toString().padLeft(2, '0');
    final monthsTr = ['Oca', 'Şub', 'Mar', 'Nis', 'May', 'Haz', 'Tem', 'Ağu', 'Eyl', 'Eki', 'Kas', 'Ara'];
    String formattedDate = "${twoDigits(model.targetDateTime.day)} ${monthsTr[model.targetDateTime.month - 1]} ${model.targetDateTime.year}";

    return Container(
      margin: const EdgeInsets.only(bottom: 14),
      padding: const EdgeInsets.symmetric(vertical: 20, horizontal: 16),
      decoration: BoxDecoration(
        gradient: isFinished
            ? const LinearGradient(colors: [Color(0xFF7F8C8D), Color(0xFF95A5A6)])
            : LinearGradient(colors: model.colors, begin: Alignment.centerLeft, end: Alignment.centerRight),
        borderRadius: BorderRadius.circular(14),
        boxShadow: [
          BoxShadow(color: Colors.black.withValues(alpha: 0.08), blurRadius: 6, offset: const Offset(0, 3)),
        ],
      ),
      child: Column(
        children: [
          Text(
            model.title.isEmpty ? 'İsimsiz' : model.title,
            style: const TextStyle(color: Colors.white, fontSize: 16, fontWeight: FontWeight.bold, letterSpacing: 0.5),
          ),
          const SizedBox(height: 16),
          isFinished
              ? const Padding(
            padding: EdgeInsets.symmetric(vertical: 6.0),
            child: Text(
              '⏱️ SÜRE DOLDU',
              style: TextStyle(color: Colors.white, fontSize: 20, fontWeight: FontWeight.bold, letterSpacing: 1),
            ),
          )
              : Row(
            mainAxisAlignment: MainAxisAlignment.spaceAround,
            children: [
              _timeSegment('GÜN', twoDigits(days)),
              _timeSegment('SAAT', twoDigits(hours)),
              _timeSegment('DAKİKA', twoDigits(minutes)),
              _timeSegment('SANİYE', twoDigits(seconds)),
            ],
          ),
          const SizedBox(height: 12),
          Text(formattedDate, style: const TextStyle(color: Colors.white70, fontSize: 11)),
        ],
      ),
    );
  }

  Widget _timeSegment(String title, String val) {
    return Column(
      children: [
        Text(title, style: const TextStyle(color: Colors.white70, fontSize: 10, fontWeight: FontWeight.w500)),
        const SizedBox(height: 4),
        Text(val, style: const TextStyle(color: Colors.white, fontSize: 32, fontWeight: FontWeight.bold)),
      ],
    );
  }
}


class ManageCountdownScreen extends StatefulWidget {
  final CountdownModel? model;
  const ManageCountdownScreen({super.key, this.model});

  @override
  State<ManageCountdownScreen> createState() => _ManageCountdownScreenState();
}

class _ManageCountdownScreenState extends State<ManageCountdownScreen> {
  final _titleController = TextEditingController();
  DateTime _date = DateTime.now().add(const Duration(days: 1));
  TimeOfDay _time = const TimeOfDay(hour: 9, minute: 0);
  int _themeIdx = 1;
  bool _isEdit = false;


  String _initialTitle = '';
  late DateTime _initialDate;
  late TimeOfDay _initialTime;
  int _initialThemeIdx = 1;

  @override
  void initState() {
    super.initState();
    if (widget.model != null) {
      _isEdit = true;
      _titleController.text = widget.model!.title;
      _date = widget.model!.targetDateTime;
      _time = TimeOfDay(hour: _date.hour, minute: _date.minute);
      _themeIdx = widget.model!.themeIndex;
    }
    _initialTitle = _titleController.text;
    _initialDate = _date;
    _initialTime = _time;
    _initialThemeIdx = _themeIdx;
  }

  DateTime get _finalDateTime => DateTime(_date.year, _date.month, _date.day, _time.hour, _time.minute);


  bool get _isChanged {
    return _titleController.text != _initialTitle ||
        _date != _initialDate ||
        _time != _initialTime ||
        _themeIdx != _initialThemeIdx;
  }

  Future<void> _selectDate() async {
    final picked = await showDatePicker(
      context: context,
      initialDate: _date,
      firstDate: DateTime(2020),
      lastDate: DateTime(2035),
    );
    if (picked != null) setState(() => _date = picked);
  }

  Future<void> _selectTime() async {
    final picked = await showTimePicker(
      context: context,
      initialTime: _time,
      builder: (context, child) {
        return MediaQuery(data: MediaQuery.of(context).copyWith(alwaysUse24HourFormat: true), child: child!);
      },
    );
    if (picked != null) setState(() => _time = picked);
  }

  void _saveForm() {
    if (_finalDateTime.isBefore(DateTime.now())) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Geri sayım için geçmiş bir tarih seçemezsiniz!'),
          backgroundColor: Colors.redAccent,
          behavior: SnackBarBehavior.floating,
        ),
      );
      return;
    }

    if (_isEdit) {
      widget.model!.title = _titleController.text;
      widget.model!.targetDateTime = _finalDateTime;
      widget.model!.themeIndex = _themeIdx;
      Navigator.pop(context, 'save');
    } else {
      Navigator.pop(context, CountdownModel(
        id: UniqueKey().toString(),
        title: _titleController.text,
        targetDateTime: _finalDateTime,
        themeIndex: _themeIdx,
        createdAt: DateTime.now(),
      ));
    }
  }


  Future<bool> _showUnsavedChangesDialog() async {
    final result = await showDialog<String>(
      context: context,
      builder: (context) => AlertDialog(
        title: const Text(
            'Değişiklikleri kaydet',
            style: TextStyle(fontWeight: FontWeight.bold, fontSize: 18)
        ),
        content: const Text(
            'Değişiklikleri kaydetmek ister misin?',
            style: TextStyle(fontSize: 15)
        ),
        actionsPadding: const EdgeInsets.symmetric(horizontal: 16, vertical: 8),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context, 'HAYIR'),
            child: const Text('HAYIR', style: TextStyle(color: Colors.blue, fontWeight: FontWeight.bold)),
          ),
          TextButton(
            onPressed: () => Navigator.pop(context, 'KAYDET'),
            child: const Text('KAYDET', style: TextStyle(color: Colors.blue, fontWeight: FontWeight.bold)),
          ),
        ],
      ),
    );

    if (result == 'KAYDET') {
      _saveForm();
      return false;
    } else if (result == 'HAYIR') {
      return true;
    }
    return false;
  }


  void _confirmDelete() {
    showDialog(
      context: context,
      builder: (context) => AlertDialog(
        title: const Text('Sayacı sil', style: TextStyle(fontWeight: FontWeight.bold)),
        content: Text('"${_titleController.text}" sayacını silmek istediğine emin misin?'),
        actions: [
          TextButton(onPressed: () => Navigator.pop(context), child: const Text('İPTAL', style: TextStyle(color: Colors.blue))),
          TextButton(
            onPressed: () {
              Navigator.pop(context);
              Navigator.pop(context, 'delete');
            },
            child: const Text('SİL', style: TextStyle(color: Colors.blue)),
          ),
        ],
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final preview = CountdownModel(
      id: 'preview',
      title: _titleController.text.isEmpty ? 'İsimsiz' : _titleController.text,
      targetDateTime: _finalDateTime,
      themeIndex: _themeIdx,
      createdAt: DateTime.now(),
    );

    final monthsEng = ['Ocak', 'Şubat', 'Mart', 'Nisan', 'Mayıs', 'Haziran', 'Temmuz', 'Ağustos', 'Eylül', 'Ekim', 'Kasım', 'Aralık'];
    String formattedDateStr = "${_date.day.toString().padLeft(2, '0')} ${monthsEng[_date.month - 1]} ${_date.year}";
    String formattedTimeStr = "${_time.hour.toString().padLeft(2, '0')}:${_time.minute.toString().padLeft(2, '0')}";

    return PopScope(
      canPop: !_isChanged,
      onPopInvokedWithResult: (didPop, result) async {
        if (didPop) return;
        final shouldPop = await _showUnsavedChangesDialog();
        if (shouldPop && context.mounted) {
          Navigator.pop(context);
        }
      },
      child: Scaffold(
        appBar: AppBar(
          title: Text(_isEdit ? 'Sayacı Düzenle' : 'Yeni Sayaç'),
          leading: IconButton(
            icon: const Icon(Icons.arrow_back),
            onPressed: () async {
              if (_isChanged) {
                final shouldPop = await _showUnsavedChangesDialog();
                if (shouldPop && context.mounted) Navigator.pop(context);
              } else {
                Navigator.pop(context);
              }
            },
          ),
          actions: [
            IconButton(
              icon: const Icon(Icons.check),
              onPressed: _saveForm,
            )
          ],
        ),
        body: SingleChildScrollView(
          padding: const EdgeInsets.all(16.0),
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              const Text('İsim', style: TextStyle(fontSize: 12, color: Colors.grey)),
              TextField(
                controller: _titleController,
                onChanged: (_) => setState(() {}),
                decoration: const InputDecoration(
                  hintText: 'örn. Final sınavı',
                  isDense: true,
                  contentPadding: EdgeInsets.symmetric(vertical: 8),
                ),
              ),
              const SizedBox(height: 24),

              const Text('Tarih', style: TextStyle(fontSize: 12, color: Colors.grey)),
              ListTile(
                contentPadding: EdgeInsets.zero,
                title: Text(formattedDateStr),
                trailing: const Icon(Icons.calendar_today, size: 20, color: Colors.teal),
                onTap: _selectDate,
              ),
              const Divider(height: 1),
              const SizedBox(height: 24),

              const Text('Saat', style: TextStyle(fontSize: 12, color: Colors.grey)),
              ListTile(
                contentPadding: EdgeInsets.zero,
